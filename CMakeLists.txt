cmake_minimum_required(VERSION 3.21)
project(symctrl VERSION 1.0.0 LANGUAGES CXX)
option(SYMCTRL_BUILD_PRODUCTION "Build Part A/B stubs and CLI" ON)
option(SYMCTRL_BUILD_TESTS "Build foundation and enabled feature tests" ON)
option(SYMCTRL_BUILD_FIXTURES "Build hand-authored fixture SDK" ON)
option(SYMCTRL_BUILD_PART2 "Enable reserved Part 2 workspace" OFF)
option(SYMCTRL_SANITIZERS "Enable address/undefined sanitizers" OFF)

add_library(symctrl_contracts INTERFACE)
target_include_directories(symctrl_contracts INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(symctrl_contracts INTERFACE cxx_std_17)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(symctrl_warnings INTERFACE)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(symctrl_warnings INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
    if(SYMCTRL_SANITIZERS)
        target_compile_options(symctrl_warnings INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(symctrl_warnings INTERFACE -fsanitize=address,undefined)
    endif()
elseif(SYMCTRL_SANITIZERS)
    message(FATAL_ERROR "SYMCTRL_SANITIZERS requires GCC or Clang")
endif()
add_subdirectory(src/common)
if(SYMCTRL_BUILD_FIXTURES)
    add_subdirectory(fixtures/support)
    add_subdirectory(examples)
endif()
if(SYMCTRL_BUILD_PRODUCTION)
    add_subdirectory(src/part_a)
    add_subdirectory(src/part_b)
    add_subdirectory(apps)
endif()
if(SYMCTRL_BUILD_PART2)
    add_subdirectory(src/part2)
endif()
if(SYMCTRL_BUILD_TESTS)
    enable_testing()
    add_library(symctrl_test_support INTERFACE)
    target_include_directories(symctrl_test_support INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/tests/support)
    target_link_libraries(symctrl_test_support INTERFACE symctrl_common symctrl_warnings)
    function(symctrl_test name target case label feature)
        add_test(NAME ${name} COMMAND ${target} ${case})
        set_tests_properties(${name} PROPERTIES LABELS "${label}" TIMEOUT 20)
        if(feature)
            set_tests_properties(${name} PROPERTIES SKIP_RETURN_CODE 77)
        endif()
    endfunction()
    add_subdirectory(tests/contracts)
    if(SYMCTRL_BUILD_PRODUCTION)
        add_subdirectory(tests/part_a)
        if(SYMCTRL_BUILD_FIXTURES)
            add_subdirectory(tests/part_b)
        endif()
        add_subdirectory(tests/integration)
    endif()
    if(SYMCTRL_BUILD_FIXTURES)
        add_subdirectory(tests/part2_handoff)
    endif()
    if(SYMCTRL_BUILD_PART2)
        add_subdirectory(tests/part2)
    endif()
endif()
