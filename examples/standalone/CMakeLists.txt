cmake_minimum_required(VERSION 3.21)
project(IndependentPart2Client LANGUAGES CXX)
get_filename_component(default_sdk "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(SYMCTRL_SDK_SOURCE "${default_sdk}" CACHE PATH "Path to shared contract checkout")
set(SYMCTRL_BUILD_PRODUCTION OFF CACHE BOOL "" FORCE)
set(SYMCTRL_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(SYMCTRL_BUILD_PART2 OFF CACHE BOOL "" FORCE)
set(SYMCTRL_BUILD_FIXTURES ON CACHE BOOL "" FORCE)
add_subdirectory("${SYMCTRL_SDK_SOURCE}" symctrl-sdk EXCLUDE_FROM_ALL)
add_executable(external_fixture_client client.cpp)
target_link_libraries(external_fixture_client PRIVATE symctrl_fixture_support)
